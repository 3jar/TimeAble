```python
"""CAT (Class–Assignment Time-constraint) 모델 — 지형 높이 계산 엔진.

docs/TIMEABLE_MODEL_V1.md 구현. 순수 함수(네트워크·저장 의존 없음).
입력은 사람 dict {classes, assignments}, 출력은 0~1.
"""
from __future__ import annotations

from datetime import datetime, timedelta

ALPHA = 1.0
BETA = 0.2
START_HOUR = 9
END_HOUR = 24
BLOCK_MINUTES = 60


def _clip01(x: float) -> float:
    return min(1.0, max(0.0, x))


def assignment_burden(d_hours: float, r_hours: float) -> float:
    """단일 과제 부담 p(d,R) = clip₀₁(α + β(1 − d/R)) = clip₀₁(1.2 − 0.2·d/R).

    d≤0(마감 도달·이후) 또는 R≤0이면 계산 대상이 아니므로 0.
    """
    if d_hours <= 0 or r_hours <= 0:
        return 0.0
    return _clip01(ALPHA + BETA * (1 - d_hours / r_hours))
```

```python
def person_burden(assignments: list[dict], t: datetime) -> float:
    """개인 과제 부담 P_i(t) = max over 유효 마감 D 의 p_i(D;t).

    유효 과제 = 미제출 & t<deadline. 마감 D 별 누적 작업시간 = Σ_{D_k≤D} R_k.
    """
    valid = [a for a in assignments if not a["submitted"] and t < a["deadline"]]
    if not valid:
        return 0.0
    best = 0.0
    for deadline in sorted({a["deadline"] for a in valid}):
        cumulative_r = sum(a["R"] for a in valid if a["deadline"] <= deadline)
        d_hours = (deadline - t).total_seconds() / 3600.0
        best = max(best, assignment_burden(d_hours, cumulative_r))
    return best
```

```python
def in_class(classes: list[dict], t: datetime) -> bool:
    """시각 t에 수업이 있나. 시작 포함·종료 제외, 요일 일치."""
    weekday = t.weekday()  # 월=0 ... 일=6
    minute = t.hour * 60 + t.minute + t.second / 60 + t.microsecond / 6e7
    return any(
        c["day"] == weekday and c["start"] <= minute < c["end"] for c in classes
    )


def constraint(person: dict, t: datetime) -> float:
    """개인 시간제약 점수 B_i(t) = C_i + (1−C_i)P_i. 수업중이면 1."""
    if in_class(person["classes"], t):
        return 1.0
    return person_burden(person["assignments"], t)
```

```python
def block_average(person: dict, block_start: datetime) -> float:
    """개인별 1시간 블록 평균 B̄_i(s). 분 가운데 60표본의 평균."""
    total = 0.0

        total += constraint(person, sample)
    return total / BLOCK_MINUTES
```

```python
def group_height(persons: list[dict], block_start: datetime) -> float:
    """그룹 블록 높이 Z(s) = 개인 블록 평균들의 평균. 0명이면 정의 안 됨."""
    if not persons:
        raise ValueError("group height undefined for zero persons")
    return sum(block_average(p, block_start) for p in persons) / len(persons)


def terrain(
    persons: list[dict],
    week_anchor: datetime,
    *,
    days: int = 5,
    start_hour: int = START_HOUR,
    end_hour: int = END_HOUR,
) -> list[list[float]]:
    """주간 지형 Z[day][block]. week_anchor = 대상 주 월요일 00:00."""
    grid: list[list[float]] = []
    for d in range(days):
        row = []
        for hour in range(start_hour, end_hour):
            block_start = week_anchor + timedelta(days=d, hours=hour)
            row.append(group_height(persons, block_start))
        grid.append(row)
    return grid
```
